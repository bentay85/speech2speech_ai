# Full runtime requirements for this Windows build (src/app.py). Install with:
#
#   uv pip install --python .venv\Scripts\python.exe -r requirements.txt
#
# CUDA 12.8 PyTorch builds are pulled from the official wheel index below;
# SageAttention and the spaCy English model are pinned to direct wheel URLs.
# Everything else comes from PyPI.
#
# Removed vs. the upstream Linux requirements.txt: nvidia-nccl-cu12,
# opencv-python-headless, gradio, xfuser, xformers, decord, flask, imageio,
# imageio-ffmpeg, pyloudnorm, easydict, ftfy, scikit-image (none are used by
# the live-streaming path).

--extra-index-url https://download.pytorch.org/whl/cu128

# Deep-learning stack. The "+cu128" local version can only be satisfied by the
# extra index above, so pip/uv cannot silently pick the CPU build from PyPI.
torch==2.7.1+cu128
torchvision==0.22.1+cu128

# Triton for torch.compile on Windows.
triton-windows==3.3.1.post21

# Accelerated attention. Prebuilt wheel for CUDA 12.8 + torch 2.7.1 from
# woct0rdho/SageAttention release v2.2.0-windows.post3. The model falls back
# to PyTorch SDPA when this is not importable, so it is safe to comment out.
sageattention @ https://github.com/woct0rdho/SageAttention/releases/download/v2.2.0-windows.post3/sageattention-2.2.0+cu128torch2.7.1.post3-cp39-abi3-win_amd64.whl

# Core numerics and imaging.
numpy
Pillow
scipy
PyYAML
opencv-python>=4.12.0.88

# Avatar model (src/flash_head).
einops
safetensors
diffusers>=0.34.0
# Pinned: 4.57.x is what the wav2vec2 feature extractor of flash_head is
# verified against; the transformers lazy importer is also not thread-safe,
# which src/app.py works around with an import lock.
transformers==4.57.3
tokenizers>=0.20.3
accelerate>=1.8.1
tqdm
mediapipe==0.10.9

# Microphone capture / speaker playback / resampling / decoding.
sounddevice>=0.5
soundfile
# Streaming sample-rate conversion for mics that reject 16 kHz (e.g. WASAPI).
soxr>=1.0
librosa

# Silero VAD v6 for streaming microphone input. CPU-only ONNX runtime; the
# model is copied to models\silero-vad\silero_vad.onnx (see README section 3.3).
silero-vad[onnx-cpu]==6.2.3

# Desktop UI shell (src/app.py). Qt Widgets only; no web engine.
PySide6-Essentials>=6.8

# Kokoro TTS (GPU via the PyTorch package). Requires eSpeak NG 1.52.0 on the
# system PATH for English out-of-dictionary fallback; see README-Windows.md.
kokoro>=0.9.4
# Xet-backed Hugging Face downloads for the Kokoro weights and voice files.
hf_xet>=1.0
# `hf` CLI used by setup.bat / the manual model-download commands in readme.md.
huggingface_hub[cli]>=0.34
# English G2P model used by misaki. It is downloaded automatically on first
# synthesis if omitted, but pinning it keeps installs reproducible/offline.
en-core-web-sm @ https://github.com/explosion/spacy-models/releases/download/en_core_web_sm-3.8.0/en_core_web_sm-3.8.0-py3-none-any.whl
